from fastapi import FastAPI, HTTPException from fastapi.middleware.cors import CORSMiddleware from pydantic import BaseModel, Field from typing import Optional from app.database import get_connection, init_db, seed_documents app = FastAPI(title="Sacred Timeline MVP") app.add_middleware( CORSMiddleware, allow_origins=["*"], allow_credentials=True, allow_methods=["*"], allow_headers=["*"], ) class DocumentInput(BaseModel): title: str religion: str tradition: Optional[str] = None language: Optional[str] = None source: Optional[str] = None source_url: Optional[str] = None category: Optional[str] = None date_start: Optional[int] = None date_end: Optional[int] = None summary: Optional[str] = None status: str = "ingested" @app.on_event("startup") def startup_event(): init_db() seed_documents() @app.get("/health") def health_check(): return {"status": "ok", "service": "sacred-timeline"} @app.get("/api/documents") def get_documents(): conn = get_connection() rows = conn.execute( """ SELECT * FROM documents ORDER BY date_start ASC, title ASC """ ).fetchall() conn.close() return [dict(row) for row in rows] @app.post("/api/documents") def create_document(payload: DocumentInput): conn = get_connection() cursor = conn.execute( """ INSERT INTO documents ( title, religion, tradition, language, source, source_url, category, date_start, date_end, summary, status ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?) """, ( payload.title, payload.religion, payload.tradition, payload.language, payload.source, payload.source_url, payload.category, payload.date_start, payload.date_end, payload.summary, payload.status, ), ) conn.commit() doc_id = cursor.lastrowid row = conn.execute("SELECT * FROM documents WHERE id = ?", (doc_id,)).fetchone() conn.close() return dict(row) @app.get("/api/timeline") def get_timeline(): conn = get_connection() rows = conn.execute( """ SELECT id, title, religion, date_start, date_end, summary FROM documents WHERE date_start IS NOT NULL OR date_end IS NOT NULL ORDER BY COALESCE(date_start, 0) ASC """ ).fetchall() conn.close() return [dict(row) for row in rows] @app.get("/api/dashboard") def dashboard_data(): conn = get_connection() documents = conn.execute( "SELECT * FROM documents ORDER BY date_start ASC, title ASC" ).fetchall() total_documents = len(documents) religions = sorted({row["religion"] for row in documents}) statuses = sorted({row["status"] for row in documents}) min_year = min((row["date_start"] for row in documents if row["date_start"] is not None), default=0) max_year = max((row["date_end"] for row in documents if row["date_end"] is not None), default=0) conn.close() return { "documents": [dict(row) for row in documents], "stats": { "total_documents": total_documents, "religions": len(religions), "traditions": len({row["tradition"] for row in documents if row["tradition"]}), "statuses": len(statuses), "earliest_year": min_year, "latest_year": max_year, }, "religion_breakdown": [ {"religion": religion, "count": sum(1 for doc in documents if doc["religion"] == religion)} for religion in religions ], } @app.get("/") def serve_dashboard(): from fastapi.responses import FileResponse return FileResponse("../frontend/index.html") @app.get("/static/{file_path:path}") def serve_static(file_path: str): return FileResponse(f"../frontend/{file_path}")
